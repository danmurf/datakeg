```go
func executeGeneratePipeline(ctx context.Context, sourcePath, outputPath, model string, trainPct, validPct, testPct, pairsPerK float64) error {
    // Step 1: Create Ollama client
    client, err := ollama.NewClient()
    if err != nil {
        return fmt.Errorf("create ollama client: %w", err)
    }

    // Step 2: Create generator with config
    gen := generator.NewGenerator(client, generator.GenerateConfig{
        Model:     model,
        TrainPct:  trainPct,
        ValidPct:  validPct,
        TestPct:   testPct,
        PairsPerK: pairsPerK,
    })

    // Step 3: Load documents from source
    docs, err := processor.LoadDocuments(sourcePath)
    if err != nil {
        return fmt.Errorf("load documents: %w", err)
    }

    if len(docs) == 0 {
        return fmt.Errorf("no .md or .txt files found in %s", sourcePath)
    }

    fmt.Printf("Found %d documents\n", len(docs))

    // Step 4: Accumulate all pairs
    var allTrain, allValid, allTest []writer.TrainingPair

    for i, doc := range docs {
        fmt.Printf("Processing [%d/%d]: %s\n", i+1, len(docs), doc.Name)

        pairs, err := gen.Generate(ctx, doc)
        if err != nil {
            return fmt.Errorf("generate pairs for %s: %w", doc.Name, err)
        }

        allTrain = append(allTrain, pairs.Train...)
        allValid = append(allValid, pairs.Valid...)
        allTest = append(allTest, pairs.Test...)

        fmt.Printf("  Generated: %d train, %d valid, %d test\n",
            len(pairs.Train), len(pairs.Valid), len(pairs.Test))
    }

    // Step 5: Ensure output directory exists
    if err := os.MkdirAll(outputPath, 0755); err != nil {
        return fmt.Errorf("create output directory: %w", err)
    }

    // Step 6: Write JSONL files
    trainFile := filepath.Join(outputPath, "train.jsonl")
    validFile := filepath.Join(outputPath, "valid.jsonl")
    testFile := filepath.Join(outputPath, "test.jsonl")

    if err := writer.WriteJSONL(trainFile, allTrain); err != nil {
        return fmt.Errorf("write train.jsonl: %w", err)
    }
    if err := writer.WriteJSONL(validFile, allValid); err != nil {
        return fmt.Errorf("write valid.jsonl: %w", err)
    }
    if err := writer.WriteJSONL(testFile, allTest); err != nil {
        return fmt.Errorf("write test.jsonl: %w", err)
    }

    // Step 7: Print summary
    fmt.Printf("\nGeneration complete!\n")
    fmt.Printf("  %s: %d pairs\n", trainFile, len(allTrain))
    fmt.Printf("  %s: %d pairs\n", validFile, len(allValid))
    fmt.Printf("  %s: %d pairs\n", testFile, len(allTest))

    return nil
}
```
